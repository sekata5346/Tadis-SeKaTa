PANGKALAN DATA DALAM TALIAN

☁️ Integrasi Google Sheets (Masa Nyata)

Semua aktiviti yang ditambah atau dikemas kini oleh guru akan disimpan terus ke dalam Google Sheets sekolah supaya dilihat oleh semua pengguna.

Panduan Pantas 3 Langkah Google Sheets:
  1. Buka Google Sheets baharu di sheets.new dan klik menu Extensions > Apps Script.
  2. Salin kod di bawah dan tampal ke dalam editor Apps Script.
  3. Klik Deploy > New deployment > Web app (Pilih Who has access: Anyone), kemudian tampal URL Web App di ruangan atas.
const SHEET_NAME = 'Takwim_Sekolah';

function doGet(e) {
  const prop = PropertiesService.getScriptProperties().getProperty('TAKWIM_DATA');
  const data = prop ? JSON.parse(prop) : { activities: [], updatedAt: null };
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const raw = e.postData.contents;
  const payload = JSON.parse(raw);
  PropertiesService.getScriptProperties().setProperty('TAKWIM_DATA', raw);
  if (payload && Array.isArray(payload.activities)) {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    if (ss) {
      let sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
      sheet.clearContents();
      sheet.appendRow(['Tarikh Mula','Tarikh Tamat','Masa Mula','Masa Tamat','Tajuk','Kategori','Unit/Panitia','Lokasi','Sasaran','Nama Penyelaras','Jawatan Penyelaras','Status','Catatan']);
      payload.activities.forEach(a => sheet.appendRow([a.date, a.endDate||a.date, a.startTime, a.endTime, a.title, a.category, a.unit, a.location, a.target, a.coordinator, a.coordinatorRole||'', a.status, a.notes||'']));
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
